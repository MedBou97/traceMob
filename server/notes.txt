-- some data about the container
docker run -d \
  --name tracemob-db \
  -p 5433:5432 \
  -e POSTGRES_USER=tracemob \
  -e POSTGRES_PASSWORD=tracemob_secret \
  -e POSTGRES_DB=tracemob \
  -v tracemob_pgdata:/var/lib/postgresql \
  mobilitydb/mobilitydb:latest


-- accessing the db inside the container
docker exec -it tracemob-db psql -U tracemob -d tracemob

-- i altered the time zone insde tracemob db to turkey with this
ALTER DATABASE tracemob SET timezone TO 'Europe/Istanbul';


-- start/stop container
docker start/stop tracemob-db
